import { useEffect } from "preact/hooks"
import kaminoidLogo from "../../assets/kaminoid.svg"
import { hostRpc } from "../../signals/host-rpc.js"
import { isWindowMaximized, sidebarVisible, sidebarMode, sidebarWidth } from "../../signals/layout.js"
import { SessionTabs } from "../session-tiles/SessionTabs.js"
import { LayoutToggles } from "./LayoutToggles.js"
import { ThemeQuickToggle } from "./ThemeQuickToggle.js"
import styles from "./Titlebar.module.css"
import { TitlebarButton } from "./TitlebarButton.js"
import { TitlebarQuickActions } from "./TitlebarQuickActions.js"

// Bridge-style titlebar: 42px tall, transparent (radial-gradient app
// background bleeds through), drag-region everywhere except the
// no-drag controls. Three clusters:
//   1. Left cluster — width pinned to sidebar width, holds brand icon
//      + quick-actions row (sidebar toggle + create/customize buttons).
//   2. Tabs slot — flex 1, holds the open-session tabs strip. Phase A
//      has zero sessions, so this is intentionally empty (a single
//      orphan "Welcome" tab read as confusing — removed).
//   3. Controls cluster — DevTools / minimize / maximize / close.
export function Titlebar() {
  useEffect(() => {
    const off = window.kamin?.window.onMaximized((m) => { isWindowMaximized.value = m })
    void window.kamin?.window.isMaximized().then((m) => { isWindowMaximized.value = m })
    return off
  }, [])

  const sidebarShown = sidebarVisible.value || sidebarMode.value === "customize"
  const leftStyle = sidebarShown
    ? { width: `${sidebarWidth.value}px` }
    : { width: "auto" }

  return (
    <header className={styles.titlebar}>
      <div className={styles.leftCluster} style={leftStyle}>
        <div className={styles.brand} aria-hidden="true">
          <img className={styles.brandLogo} src={kaminoidLogo} alt="" draggable={false} />
        </div>
        <TitlebarQuickActions />
      </div>
      <div className={styles.tabsSlot} aria-label="Open sessions"><SessionTabs /></div>

      <button
        className={styles.searchButton}
        aria-label="Open command palette (Ctrl+Shift+P)"
        data-tooltip="Open command palette (Ctrl+Shift+P)"
        onClick={() => { void hostRpc.commands.execute("workbench.action.showCommands") }}
      >
        <span className="codicon codicon-search" />
        <span className={styles.searchHint}>Type a command…</span>
      </button>
      <LayoutToggles />
      <ThemeQuickToggle />
      <div className={styles.controls}>
        <TitlebarButton
          iconSet="fas"
          icon="fa-bug"
          variant="devtools"
          label="DevTools"
          onClick={() => { void window.kamin?.window.toggleDevTools() }}
        />
        {/* Window controls use codicons' chrome-* set — purpose-built
            for this exact role and visually match Windows/macOS native
            conventions. FontAwesome's `fa-square` solid (filled) read
            wrong as a maximize affordance. */}
        <TitlebarButton
          iconSet="codicon"
          icon="chrome-minimize"
          label="Minimize"
          onClick={() => { void window.kamin?.window.minimize() }}
        />
        <TitlebarButton
          iconSet="codicon"
          icon={isWindowMaximized.value ? "chrome-restore" : "chrome-maximize"}
          label={isWindowMaximized.value ? "Restore" : "Maximize"}
          onClick={() => { void window.kamin?.window.maximize() }}
        />
        <TitlebarButton
          iconSet="codicon"
          icon="chrome-close"
          variant="close"
          label="Close"
          onClick={() => { void window.kamin?.window.close() }}
        />
      </div>
    </header>
  )
}
